"""
다중 AI 프로바이더 지원 모듈
- 로컬: Ollama, LM Studio
- 클라우드: OpenAI, Anthropic, Google Gemini, Groq

import logging
import os
import urllib.request
import urllib.error
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional

logger = logging.getLogger(__name__)


# ============================================================
# 설정 데이터 클래스
# ============================================================

@dataclass
class AIConfig:
    """AI 프로바이더 설정"""
    provider: str = "ollama"  # ollama, lmstudio, openai, anthropic, google, groq
    model: str = "qwen2.5:7b"
    api_key: Optional[str] = None
    api_base: Optional[str] = None
    timeout: int = 30
    temperature: float = 0.1
    max_tokens: int = 100


# ============================================================
# 기본 프로바이더 클래스
# ============================================================

class AIProvider(ABC):
    """AI 프로바이더 추상 기본 클래스"""

    def __init__(self, config: AIConfig):
        self.config = config

    @abstractmethod
    def generate(self, prompt: str) -> Optional[str]:
        """프롬프트에 대한 응답 생성"""
        pass

    @abstractmethod
    def is_available(self) -> bool:
        """서비스 사용 가능 여부 확인"""
        pass

    def _make_request(
        self,
        url: str,
        data: dict,
        headers: dict
    ) -> Optional[dict]:
        """HTTP 요청 공통 처리"""
        try:
            request_data = json.dumps(data).encode("utf-8")
            req = urllib.request.Request(
                url,
                data=request_data,
                headers=headers,
                method="POST"
            )
            with urllib.request.urlopen(req, timeout=self.config.timeout) as response:
                return json.loads(response.read().decode("utf-8"))
        except urllib.error.URLError as e:
            logger.warning(f"요청 실패: {e}")
            return None
        except Exception as e:
            logger.warning(f"처리 오류: {e}")
            return None

# Ollama 프로바이더

class OllamaProvider(AIProvider):
    """Ollama 로컬 AI 프로바이더"""

    def __init__(self, config: AIConfig):
        super().__init__(config)
        self.base_url = config.api_base or "http://localhost:11434"

    def generate(self, prompt: str) -> Optional[str]:
        data = {
            "model": self.config.model,
            "prompt": prompt,
            "stream": False,
            "options": {
                "temperature": self.config.temperature,
                "num_predict": self.config.max_tokens
            }
        }
        headers = {"Content-Type": "application/json"}

        result = self._make_request(
            f"{self.base_url}/api/generate",
            data,
            headers
        )

        if result:
            return result.get("response", "").strip()
        return None

    def is_available(self) -> bool:
        try:
            req = urllib.request.Request(
                f"{self.base_url}/api/tags",
                method="GET"
            )
            with urllib.request.urlopen(req, timeout=5) as response:
                result = json.loads(response.read().decode("utf-8"))
                models = [m.get("name", "") for m in result.get("models", [])]
                model_base = self.config.model.split(":")[0]
                return any(
                    m == self.config.model or m.startswith(f"{model_base}:")
                    for m in models
                )
        except Exception:
            return False

class LMStudioProvider(AIProvider):

        super().__init__(config)

            "temperature": self.config.temperature,
            "max_tokens": self.config.max_tokens
        }
        headers = {"Content-Type": "application/json"}

        result = self._make_request(
            f"{self.base_url}/chat/completions",
            data,
            headers
        )

        if result and "choices" in result:
            return result["choices"][0]["message"]["content"].strip()
        return None

    def is_available(self) -> bool:
        try:
            req = urllib.request.Request(
                f"{self.base_url}/models",
                method="GET"
            )
            with urllib.request.urlopen(req, timeout=5) as response:
                return response.status == 200
        except Exception:
            return False

class GoogleGeminiProvider(AIProvider):
    """Google Gemini API 프로바이더"""

    def __init__(self, config: AIConfig):
        super().__init__(config)
        self.api_key = config.api_key or os.getenv("GOOGLE_API_KEY")
        self.model = config.model or "gemini-1.5-flash"

        if not self.api_key:
            logger.warning("Google API 키가 설정되지 않았습니다.")

    def generate(self, prompt: str) -> Optional[str]:
        if not self.api_key:
            return None

        url = (
            f"https://generativelanguage.googleapis.com/v1beta/models/"
            f"{self.model}:generateContent?key={self.api_key}"
        )

        data = {
            "contents": [{"parts": [{"text": prompt}]}],
            "generationConfig": {
                "temperature": self.config.temperature,
                "maxOutputTokens": self.config.max_tokens
            }
        }
        headers = {"Content-Type": "application/json"}

        result = self._make_request(url, data, headers)

        if result and "candidates" in result:
            try:
                return result["candidates"][0]["content"]["parts"][0]["text"].strip()
            except (KeyError, IndexError):
                pass
        return None

    def is_available(self) -> bool:
        return bool(self.api_key)

PROVIDERS = {
    "ollama": OllamaProvider,
    "lmstudio": LMStudioProvider,
    "openai": OpenAIProvider,
    "anthropic": AnthropicProvider,
    "google": GoogleGeminiProvider,
    "gemini": GoogleGeminiProvider,

def create_provider(config: AIConfig) -> AIProvider:
    """설정에 따라 적절한 AI 프로바이더 생성"""
    provider_class = PROVIDERS.get(config.provider.lower())

    if not provider_class:
        raise ValueError(
            f"지원하지 않는 프로바이더: {config.provider}\n"
            f"사용 가능: {list(PROVIDERS.keys())}"
        )

    return provider_class(config)


def get_available_provider(preferred: list[str] = None) -> Optional[AIProvider]:
    """
    사용 가능한 프로바이더를 자동 탐지하여 반환합니다.

    Args:
        preferred: 우선 순위 프로바이더 목록

    Returns:
        사용 가능한 첫 번째 프로바이더
    """
    if preferred is None:
        preferred = ["ollama", "lmstudio", "groq", "google", "openai", "anthropic"]

    for provider_name in preferred:
        try:
            config = AIConfig(provider=provider_name)
            provider = create_provider(config)
            if provider.is_available():
                logger.info(f"사용 가능한 프로바이더 발견: {provider_name}")
                return provider
        except Exception as e:
            logger.debug(f"{provider_name} 확인 실패: {e}")
            continue

    logger.warning("사용 가능한 AI 프로바이더가 없습니다.")
    return None

# 북마크 매칭 통합 함수
# ============================================================

def match_bookmark_with_ai(
    title: str,
    url: str,
    existing_folders: list[str],
    provider: AIProvider
) -> Optional[str]:
    """
    AI를 사용하여 북마크를 기존 폴더 중 하나에 매칭합니다.

    Args:
        title: 북마크 제목
        url: 북마크 URL
        existing_folders: 기존 폴더 목록
        provider: AI 프로바이더

    Returns:
        매칭된 폴더 경로 또는 None
    """
    if not existing_folders or not provider:
        return None

    folder_list = "\n".join(
        f"{i+1}. {folder}" for i, folder in enumerate(existing_folders)
    )

    prompt = f"""다음 북마크를 아래 폴더 목록 중 가장 적절한 폴더에 분류해주세요.

북마크 정보:
- 제목: {title}
- URL: {url}

사용 가능한 폴더 목록:
{folder_list}

규칙:
1. 반드시 위 목록에 있는 폴더 중 하나만 선택하세요.
2. 새로운 폴더를 만들지 마세요.
3. 폴더 이름만 정확히 출력하세요 (번호 제외).
4. 적절한 폴더가 없으면 "없음"이라고 답하세요.

답변:"""

    try:
        answer = provider.generate(prompt)

        if answer and answer != "없음":
            # 정확히 일치하는 폴더 찾기
            for folder in existing_folders:
                if folder == answer or folder.endswith(f"/{answer}"):
                    return folder

            # 부분 일치 시도
            answer_lower = answer.lower()
            for folder in existing_folders:
                if answer_lower in folder.lower():
                    return folder

        return None

    except Exception as e:
        logger.warning(f"AI 매칭 오류: {e}")
        return None

def normalize_url(raw_url: str) -> str:
    """
    URL을 정규화합니다.
    - 프로토콜 및 호스트 소문자화
    - 기본 포트 제거
    - 추적 파라미터 제거
    - 후행 슬래시 정리
    """
    try:
        url = raw_url.strip()
        if not url:
            return ""

        parts = urlsplit(url)
        scheme = parts.scheme.lower()
        netloc = parts.netloc.lower()

        # 기본 포트 제거
        if netloc.endswith(":80") and scheme == "http":
            netloc = netloc[:-3]
        elif netloc.endswith(":443") and scheme == "https":
            netloc = netloc[:-4]

        # www 접두사 통일 (선택적)
        # if netloc.startswith("www."):
        #     netloc = netloc[4:]

        # 추적 파라미터 제거
        filtered_query = [
            (k, v) for k, v in parse_qsl(parts.query, keep_blank_values=True)
            if k.lower() not in TRACKING_PARAMS
        ]
        query = urlencode(filtered_query)

        # 경로 정리
        path = parts.path or "/"
        if path != "/" and path.endswith("/"):
            path = path.rstrip("/")

        # 프래그먼트 제거 (선택적)
        fragment = ""  # parts.fragment를 유지하려면 이 줄 수정

        return urlunsplit((scheme, netloc, path, query, fragment))

    except Exception as e:
        logger.warning(f"URL 정규화 실패: {raw_url} - {e}")
        return raw_url


def is_valid_url(url: str) -> bool:
    """URL이 유효한 형식인지 확인합니다."""
    if not url:
        return False
    try:
        parts = urlsplit(url)
        return parts.scheme in ("http", "https") and bool(parts.netloc)
    except Exception:
        return False

    북마크 HTML에서 모든 폴더 경로를 재귀적으로 수집합니다.
    """
    folders = []
    if not dl_tag:
        return folders

    for dt in dl_tag.find_all("dt", recursive=False):
        h3 = dt.find("h3", recursive=False)
        sub_dl = dt.find("dl", recursive=False)

        if h3 and sub_dl:
            folder_name = h3.get_text(strip=True)
            full_path = f"{prefix}/{folder_name}" if prefix else folder_name
            folders.append(full_path)
            folders.extend(collect_folders(sub_dl, full_path))

    return folders


def get_current_folder_path(a_tag: Tag) -> str:
    """
    북마크 링크가 속한 현재 폴더 경로를 반환합니다.
    """
    path_parts = []
    current = a_tag.parent

    while current:
        if current.name == "dl":
            # 이 DL의 부모 DT에서 H3를 찾음
            parent_dt = current.find_parent("dt")
            if parent_dt:
                h3 = parent_dt.find("h3", recursive=False)
                if h3:
                    path_parts.insert(0, h3.get_text(strip=True))
        current = current.parent

    return "/".join(path_parts) if path_parts else "루트"


def find_or_create_folder_dl(soup: BeautifulSoup, root_dl: Tag, folder_path: str) -> Tag:
    """
    지정된 폴더 경로에 해당하는 DL 태그를 찾습니다.
    (새 폴더 생성은 차단되어 있으므로 기존 폴더만 검색)
    """
    if not folder_path or folder_path == "루트":
        return root_dl

    parts = folder_path.split("/")
    current_dl = root_dl

    for part in parts:
        found = False
        for dt in current_dl.find_all("dt", recursive=False):
            h3 = dt.find("h3", recursive=False)
            if h3 and h3.get_text(strip=True) == part:
                sub_dl = dt.find("dl", recursive=False)
                if sub_dl:
                    current_dl = sub_dl
                    found = True
                    break

        if not found:
            logger.warning(f"폴더를 찾을 수 없음: {folder_path}")
            return root_dl

    return current_dl

def load_rules(rules_path: Path) -> list[dict]:
    """규칙 파일을 로드합니다."""
    if not rules_path.exists():
        logger.warning(f"규칙 파일이 없습니다: {rules_path}")
        return []

    try:
        rules = json.loads(rules_path.read_text(encoding="utf-8"))
        logger.info(f"규칙 {len(rules)}개 로드됨")
        return rules
    except json.JSONDecodeError as e:
        logger.error(f"규칙 파일 파싱 오류: {e}")
        return []


def match_by_rules(
    title: str,
    url: str,
    rules: list[dict],
    valid_folders: set[str]
) -> Optional[str]:
    """
    규칙 기반으로 북마크를 폴더에 매칭합니다.
    """
    haystack = f"{title} {url}".lower()
    host = urlsplit(url).netloc.lower()

    for rule in rules:
        target = rule.get("folder", "")

        # 유효한 폴더인지 확인
        if target not in valid_folders:
            continue

        keywords = [kw.lower() for kw in rule.get("keywords", [])]
        domains = [dm.lower() for dm in rule.get("domains", [])]

        # 도메인 매칭
        if any(dm in host for dm in domains):
            return target

        # 키워드 매칭
        if any(kw in haystack for kw in keywords):
            return target

    return None

def match_by_ollama(
    title: str,
    url: str,
    existing_folders: list[str],
    model: str = "qwen2.5:7b"
) -> Optional[str]:
    """
    로컬 Ollama를 사용하여 북마크를 기존 폴더 중 하나에 매칭합니다.
    새 폴더 생성은 허용하지 않습니다.
    """
    if not existing_folders:
        return None

    # 폴더 목록을 번호와 함께 포맷팅
    folder_list = "\n".join(
        f"{i+1}. {folder}" for i, folder in enumerate(existing_folders)
    )

    prompt = f"""다음 북마크를 아래 폴더 목록 중 가장 적절한 폴더에 분류해주세요.

북마크 정보:
- 제목: {title}
- URL: {url}

사용 가능한 폴더 목록:
{folder_list}

규칙:
1. 반드시 위 목록에 있는 폴더 중 하나만 선택하세요.
2. 새로운 폴더를 만들지 마세요.
3. 폴더 이름만 정확히 출력하세요 (번호 제외).
4. 적절한 폴더가 없으면 "없음"이라고 답하세요.

답변:"""

    try:
        request_data = json.dumps({
            "model": model,
            "prompt": prompt,
            "stream": False,
            "options": {
                "temperature": 0.1,
                "num_predict": 100
            }
        }).encode("utf-8")

        req = urllib.request.Request(
            OLLAMA_API_URL,
            data=request_data,
            headers={"Content-Type": "application/json"},
            method="POST"
        )

        with urllib.request.urlopen(req, timeout=OLLAMA_TIMEOUT) as response:
            result = json.loads(response.read().decode("utf-8"))
            answer = result.get("response", "").strip()

            # 응답 검증: 기존 폴더 목록에 있는지 확인
            if answer and answer != "없음":
                # 정확히 일치하는 폴더 찾기
                for folder in existing_folders:
                    if folder == answer or folder.endswith(f"/{answer}"):
                        logger.debug(f"AI 매칭 성공: {title[:30]}... → {folder}")
                        return folder

                # 부분 일치 시도
                answer_lower = answer.lower()
                for folder in existing_folders:
                    if answer_lower in folder.lower():
                        logger.debug(f"AI 부분 매칭: {title[:30]}... → {folder}")
                        return folder

            return None

    except urllib.error.URLError as e:
        logger.warning(f"Ollama 연결 실패: {e}")
        return None
    except Exception as e:
        logger.warning(f"Ollama 처리 오류: {e}")
        return None


def check_ollama_available(model: str = "qwen2.5:7b") -> bool:
    """Ollama 서버와 모델이 사용 가능한지 확인합니다."""
    try:
        req = urllib.request.Request(
            "http://localhost:11434/api/tags",
            method="GET"
        )
        with urllib.request.urlopen(req, timeout=5) as response:
            result = json.loads(response.read().decode("utf-8"))
            models = [m.get("name", "") for m in result.get("models", [])]
            
            # 모델 이름 매칭 (버전 태그 포함/미포함 모두 확인)
            model_base = model.split(":")[0]
            available = any(
                m == model or m.startswith(f"{model_base}:")
                for m in models
            )
            
            if available:
                logger.info(f"Ollama 모델 확인됨: {model}")
            else:
                logger.warning(f"Ollama 모델 없음: {model} (사용 가능: {models})")
            
            return available

    except Exception as e:
        logger.warning(f"Ollama 서버 확인 실패: {e}")
        return False

def move_bookmark_to_folder(
    soup: BeautifulSoup,
    root_dl: Tag,
    a_tag: Tag,
    target_folder: str,
    stats: ProcessingStats
) -> bool:
    """
    북마크를 지정된 폴더로 실제 이동합니다.
    """
    try:
        # 현재 위치 확인
        current_folder = get_current_folder_path(a_tag)
        if current_folder == target_folder:
            return False  # 이미 올바른 위치

        # 대상 폴더 DL 찾기
        target_dl = find_or_create_folder_dl(soup, root_dl, target_folder)
        if target_dl == root_dl and target_folder and target_folder != "루트":
            logger.warning(f"대상 폴더를 찾을 수 없음: {target_folder}")
            return False

        # 현재 DT 요소 가져오기
        current_dt = a_tag.find_parent("dt")
        if not current_dt:
            return False

        # DT 요소 복사 및 이동
        new_dt = copy(current_dt)
        target_dl.append(new_dt)

        # 원본 제거
        current_dt.decompose()

        # 통계 기록
        stats.moved_items += 1
        stats.move_details.append({
            "title": a_tag.get_text(strip=True)[:50],
            "from": current_folder,
            "to": target_folder
        })

        return True

    except Exception as e:
        logger.error(f"북마크 이동 실패: {e}")
        stats.errors += 1
        return False

def process_bookmarks(
    input_html: Path,
    output_html: Path,
    rules_path: Path,
    ai_config: Optional[AIConfig] = None,  # 변경됨
    actually_move: bool = False,
    report_path: Optional[Path] = None
) -> ProcessingStats:
    """
    북마크 HTML 파일을 처리합니다.
    """

    Args:
        input_html: 입력 북마크 HTML 파일 경로
        output_html: 출력 북마크 HTML 파일 경로
        rules_path: 규칙 JSON 파일 경로
        use_ollama: Ollama AI 매칭 사용 여부
        ollama_model: 사용할 Ollama 모델명
        actually_move: 실제로 북마크를 이동할지 여부
        report_path: 상세 리포트 저장 경로 (선택)

    Returns:
        ProcessingStats: 처리 통계
    """
    logger.info(f"처리 시작: {input_html}")
    stats = ProcessingStats()

    # 입력 파일 읽기
    try:
        html_text = input_html.read_text(encoding="utf-8", errors="ignore")
    except FileNotFoundError:
        logger.error(f"입력 파일을 찾을 수 없습니다: {input_html}")
        raise

    # HTML 파싱
    soup = BeautifulSoup(html_text, "html.parser")
    root_dl = soup.find("dl")
    if not root_dl:
        raise ValueError("유효한 북마크 DL 구조를 찾을 수 없습니다.")

    # 규칙 로드
    rules = load_rules(rules_path)

    # 기존 폴더 목록 수집
    existing_folders = collect_folders(root_dl)
    valid_folder_set = set(existing_folders)
    logger.info(f"기존 폴더 {len(existing_folders)}개 발견")

    # AI 프로바이더 설정
    ai_provider = None
    if ai_config:
        try:
            ai_provider = create_provider(ai_config)
            if not ai_provider.is_available():
                logger.warning(f"{ai_config.provider} 사용 불가, 규칙 기반만 수행")
                ai_provider = None
            else:
                logger.info(f"AI 프로바이더 활성화: {ai_config.provider}")
        except Exception as e:
            logger.warning(f"AI 프로바이더 생성 실패: {e}")

    # 중복 확인용 집합
    seen_urls: dict[str, BookmarkItem] = {}

    # 처리할 북마크 수집 (먼저 수집 후 처리 - 순회 중 수정 방지)
    bookmarks_to_process: list[tuple[Tag, BookmarkItem]] = []

    for a_tag in root_dl.find_all("a"):
        raw_href = a_tag.get("href", "")

        # HTTP(S) URL만 처리
        if not is_valid_url(raw_href):
            continue

        stats.total_links += 1
        norm_url = normalize_url(raw_href)
        title = a_tag.get_text(strip=True)
        current_folder = get_current_folder_path(a_tag)

        # 북마크 항목 생성
        bookmark = BookmarkItem(
            title=title,
            url=raw_href,
            normalized_url=norm_url,
            original_folder=current_folder,
            add_date=a_tag.get("add_date"),
            icon=a_tag.get("icon")
        )

        # 중복 확인
        if norm_url in seen_urls:
            stats.duplicates_removed += 1
            stats.duplicate_details.append({
                "kept": seen_urls[norm_url].title[:50],
                "removed": title[:50],
                "url": norm_url[:80]
            })
            # 중복 항목 제거
            parent_dt = a_tag.find_parent("dt")
            if parent_dt:
                parent_dt.decompose()
            continue

        seen_urls[norm_url] = bookmark
        bookmarks_to_process.append((a_tag, bookmark))

    logger.info(f"처리 대상 북마크: {len(bookmarks_to_process)}개")

    # 북마크 매칭 및 이동 처리
    for a_tag, bookmark in bookmarks_to_process:
        # 1차: 규칙 기반 매칭
        matched_folder = match_by_rules(
            bookmark.title,
            bookmark.url,
            rules,
            valid_folder_set
        )

        if matched_folder:
            stats.matched_by_rule += 1
            bookmark.suggested_folder = matched_folder
            bookmark.match_method = "rule"

        # 2차: AI 매칭
        elif ai_provider:
            ai_folder = match_bookmark_with_ai(
                bookmark.title,
                bookmark.url,
                existing_folders,
                ai_provider  # 변경됨
            )
            if ai_folder:
                stats.matched_by_ai += 1
                bookmark.suggested_folder = ai_folder
                bookmark.match_method = "ai"
            else:
                stats.unmatched += 1
                bookmark.match_method = "unmatched"
        else:
            stats.unmatched += 1
            bookmark.match_method = "unmatched"

        # 속성 기록 (검토용)
        if bookmark.suggested_folder:
            a_tag["data-suggested-folder"] = bookmark.suggested_folder
            a_tag["data-match-method"] = bookmark.match_method

        # 실제 이동 수행
        if actually_move and bookmark.suggested_folder:
            move_bookmark_to_folder(
                soup, root_dl, a_tag,
                bookmark.suggested_folder, stats
            )

    # 결과 저장
    output_html.write_text(str(soup), encoding="utf-8")
    logger.info(f"결과 저장됨: {output_html}")

    # 상세 리포트 저장
    if report_path:
        report = {
            "처리 일시": datetime.now().isoformat(),
            "입력 파일": str(input_html),
            "출력 파일": str(output_html),
            "통계": stats.to_dict(),
            "중복 상세": stats.duplicate_details[:100],  # 최대 100개
            "이동 상세": stats.move_details[:100]
        }
        report_path.write_text(
            json.dumps(report, indent=2, ensure_ascii=False),
            encoding="utf-8"
        )
        logger.info(f"리포트 저장됨: {report_path}")

    return stats

## 5. 사용 방법

### 5.1 기본 실행 (규칙 기반만 사용, 속성 기록만)

```bash
python bookmark_organizer.py
```

### 5.2 로컬 AI 사용

```bash
# Ollama
python bookmark_organizer.py --ai-provider ollama --ai-model qwen2.5:7b

# LM Studio
python bookmark_organizer.py --ai-provider lmstudio --ai-model local-model

# 자동 탐지 (로컬 우선)
python bookmark_organizer.py --ai-provider auto
```

### 5.3 클라우드 API 사용

```bash
# OpenAI
export OPENAI_API_KEY="sk-..."
python bookmark_organizer.py --ai-provider openai --ai-model gpt-4o-mini

# Anthropic Claude
export ANTHROPIC_API_KEY="sk-ant-..."
python bookmark_organizer.py --ai-provider anthropic --ai-model claude-3-5-sonnet-20241022

# Google Gemini (무료 티어)
export GOOGLE_API_KEY="..."
python bookmark_organizer.py --ai-provider google --ai-model gemini-1.5-flash

# Groq (빠른 추론, 무료 티어)
export GROQ_API_KEY="gsk_..."
python bookmark_organizer.py --ai-provider groq --ai-model llama-3.1-8b-instant
```

### 5.4 실제 이동 수행 + 리포트 생성

```bash
python bookmark_organizer.py --use-ollama --move --report report.json
```

### 5.5 전체 옵션

```bash
python bookmark_organizer.py \
    -i bookmarks_backup.html \
    -o bookmarks_cleaned.html \
    -r rules.json \
    --use-ollama \
    --ollama-model llama3.1:8b \
    --move \
    --report processing_report.json
```

## 7. 추가 권장 사항

1. **테스트 환경에서 먼저 실행**: 실제 북마크에 적용하기 전에 별도 Chrome 프로필에서 결과를 검증하시기 바랍니다.

2. **규칙 파일 점진적 확장**: 처음에는 간단한 규칙으로 시작하고, 미분류된 항목을 분석하여 규칙을 추가하는 것이 좋습니다.

3. **백업 필수**: 스크립트 실행 전 원본 `bookmarks_backup.html`을 별도로 보관해 두세요.
